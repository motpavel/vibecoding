const p = {

    days:    n => [,'день','дня','дней'][n%10===1&&n%100!==11?1:[2,3,4].includes(n%10)&&![12,13,14].includes(n%100)?2:3],

    hours:   n => [,'час','часа','часов'][n%10===1&&n%100!==11?1:[2,3,4].includes(n%10)&&![12,13,14].includes(n%100)?2:3],

    minutes: n => [,'минута','минуты','минут'][n%10===1&&n%100!==11?1:[2,3,4].includes(n%10)&&![12,13,14].includes(n%100)?2:3],

    seconds: n => [,'секунда','секунды','секунд'][n%10===1&&n%100!==11?1:[2,3,4].includes(n%10)&&![12,13,14].includes(n%100)?2:3],

};


function formatCountdown(ms) {

    if (ms <= 0) return 'начинается!';

    const s   = Math.floor(ms / 1000);

    const sec = s % 60;

    const min = Math.floor(s / 60) % 60;

    const hrs = Math.floor(s / 3600) % 24;

    const day = Math.floor(s / 86400);

    const totalMin = Math.floor(s / 60);

    const totalHrs = s / 3600;


    if (totalMin < 10) {

        return `${min} ${p.minutes(min)} ${sec} ${p.seconds(sec)}`;

    } else if (totalHrs < 3) {

        return hrs > 0

            ? `${hrs} ${p.hours(hrs)} ${min} ${p.minutes(min)}`

            : `${min} ${p.minutes(min)}`;

    } else if (day > 0) {

        return `${day} ${p.days(day)} ${hrs} ${p.hours(hrs)}`;

    } else {

        return `${hrs} ${p.hours(hrs)} ${min} ${p.minutes(min)}`;

    }

}


function formatElapsed(ms) {

    const s   = Math.floor(ms / 1000);

    const min = Math.floor(s / 60) % 60;

    const hrs = Math.floor(s / 3600);

    if (hrs > 0) return `${hrs} ${p.hours(hrs)} ${min} ${p.minutes(min)}`;

    return `${min} ${p.minutes(min)}`;

}


// ── Timer loop ────────────────────────────────────────

function tick() {

    const now = new Date();


    // Active (elapsed)

    document.getElementById('timer-active').textContent = formatElapsed(now - activeStart);


    // Far

    const dFar = targetFar - now;

    document.getElementById('timer-far').textContent = formatCountdown(dFar);


    // Mid

    const dMid = targetMid - now;

    const cdMid = document.getElementById('cd-mid');

    document.getElementById('timer-mid').textContent = formatCountdown(dMid);

    const midMin = Math.floor(dMid / 1000 / 60);

    cdMid.className = 'countdown ' + (midMin < 10 ? 'cd-urgent' : '');

    cdMid.querySelector('.cd-icon').textContent = midMin < 10 ? '🔴' : '⏳';


    // Near

    const dNear = targetNear - now;

    document.getElementById('timer-near').textContent = formatCountdown(dNear);

}


tick();

setInterval(tick, 1000);


// ── Popup ─────────────────────────────────────────────

function openPopup(key) {

    const d = teams[key];

    document.getElementById('pop-subtitle').textContent = d.game;

    document.getElementById('pop-title').textContent = `Участвующие команды`;

    document.getElementById('pop-list').innerHTML = d.list

        .map((t, i) => `<li class="team-item"><div class="team-num">${i+1}</div><div class="team-name-text">${t}</div></li>`)

        .join('');

    document.getElementById('overlay').classList.add('open');

}


function closePopup() {

    document.getElementById('overlay').classList.remove('open');

}


function overlayClick(e) {

    if (e.target === document.getElementById('overlay')) closePopup();

}
